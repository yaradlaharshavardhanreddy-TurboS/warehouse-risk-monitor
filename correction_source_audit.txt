"""Generate deterministic warehouse door activity for Phase 4.

Run:
    python -m app.generator.door_event_data
    python -m app.generator.door_event_data --replace
"""

import argparse
import random
import sqlite3
from datetime import datetime, timedelta

from app.database.database import DATABASE_PATH
from scripts.phase10_generation_config import get_generator_seed

SEED = get_generator_seed('door_events')
SIMULATION_START = datetime(2026, 10, 22, 0, 0, 0)
SIMULATION_DAYS = 7


def _load_zones(connection):
    rows = connection.execute(
        """
        SELECT zone_id, zone_type
        FROM zones
        WHERE lifecycle_state = 'ACTIVE'
        ORDER BY zone_id
        """
    ).fetchall()

    if len(rows) != 100:
        raise RuntimeError(
            f"Expected 100 active zones; found {len(rows)}."
        )
    return rows


def generate_door_events(replace=False):
    rng = random.Random(SEED)
    connection = sqlite3.connect(DATABASE_PATH)
    connection.execute("PRAGMA foreign_keys = ON")

    try:
        existing = connection.execute(
            "SELECT COUNT(*) FROM door_events"
        ).fetchone()[0]

        if existing and not replace:
            raise RuntimeError(
                f"door_events already contains {existing:,} rows. "
                "Use --replace only if you intentionally want to regenerate it."
            )

        zones = _load_zones(connection)
        if replace:
            connection.execute("DELETE FROM door_events")

        simulation_end = SIMULATION_START + timedelta(days=SIMULATION_DAYS)
        event_records = []
        event_number = 0

        for zone_id, zone_type in zones:
            # Loading bays are busier; freezer/cold doors are opened less often.
            if zone_type == "LOADING":
                interval_range = (35, 120)
                duration_range = (30, 600)
            elif zone_type in ("FREEZER", "COLD"):
                interval_range = (100, 260)
                duration_range = (10, 150)
            else:
                interval_range = (60, 190)
                duration_range = (15, 240)

            current = SIMULATION_START + timedelta(
                minutes=rng.randint(0, 45)
            )
            door_identifier = f"DOOR-{zone_id}-01"

            while current < simulation_end:
                duration = rng.randint(*duration_range)

                # A small fraction of cycles are abnormal, while retaining
                # a coherent open-to-close sequence.
                roll = rng.random()
                if roll < 0.005:
                    opening_type = "FORCED_OPEN"
                else:
                    opening_type = "OPENED"

                event_number += 1
                event_records.append(
                    (
                        f"DR-{event_number:08d}",
                        zone_id,
                        door_identifier,
                        opening_type,
                        current.strftime("%Y-%m-%d %H:%M:%S"),
                        None,
                        "SENSOR",
                    )
                )

                if roll >= 0.005 and roll < 0.025 and duration >= 60:
                    held_at = current + timedelta(seconds=min(60, duration // 2))
                    event_number += 1
                    event_records.append(
                        (
                            f"DR-{event_number:08d}",
                            zone_id,
                            door_identifier,
                            "HELD_OPEN",
                            held_at.strftime("%Y-%m-%d %H:%M:%S"),
                            int((held_at - current).total_seconds()),
                            "SENSOR",
                        )
                    )

                closed_at = current + timedelta(seconds=duration)
                event_number += 1
                event_records.append(
                    (
                        f"DR-{event_number:08d}",
                        zone_id,
                        door_identifier,
                        "CLOSED",
                        closed_at.strftime("%Y-%m-%d %H:%M:%S"),
                        duration,
                        "SENSOR",
                    )
                )

                current += timedelta(
                    minutes=rng.randint(*interval_range)
                )

        connection.executemany(
            """
            INSERT INTO door_events (
                door_event_id, zone_id, door_identifier, event_type,
                event_timestamp, duration_seconds, recorded_by
            ) VALUES (?, ?, ?, ?, ?, ?, ?)
            """,
            event_records,
        )

        actual = connection.execute(
            "SELECT COUNT(*) FROM door_events"
        ).fetchone()[0]
        if actual != len(event_records):
            raise RuntimeError(
                f"Expected {len(event_records):,} door events; found {actual:,}."
            )

        connection.commit()

        print("Door event generation successful!")
        print("Active zones:", len(zones))
        print("Door identifiers:", len(zones))
        print("Door events inserted:", f"{len(event_records):,}")
        print("Simulation start:", SIMULATION_START.isoformat(sep=" "))
        print("Simulation days:", SIMULATION_DAYS)
        print("Event type distribution:")
        for event_type, count in connection.execute(
            """
            SELECT event_type, COUNT(*)
            FROM door_events
            GROUP BY event_type
            ORDER BY event_type
            """
        ):
            print(f"  {event_type}: {count:,}")
        print("Random seed:", SEED)

    except Exception:
        connection.rollback()
        raise
    finally:
        connection.close()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--replace", action="store_true",
        help="Delete existing door events before generating new ones.",
    )
    args = parser.parse_args()
    generate_door_events(args.replace)


if __name__ == "__main__":
    main()

"""Generate inventory lots and their zone assignments."""

import random
import sqlite3
from datetime import datetime, timedelta

from app.database.database import DATABASE_PATH
from scripts.phase10_generation_config import get_generator_seed

SEED = get_generator_seed('inventory_lots')
LOT_COUNT = 10_000

BASE_DATE = datetime(2026, 1, 1)

PRODUCT_ZONE_MAPPING = {
    "DAIRY": "COLD",
    "PRODUCE": "COLD",
    "FROZEN_FOOD": "FREEZER",
    "PHARMA": "COLD",
    "ELECTRONICS": "AMBIENT",
}


def format_timestamp(value):
    return value.strftime("%Y-%m-%d %H:%M:%S")


def generate_inventory():
    rng = random.Random(SEED)

    connection = sqlite3.connect(DATABASE_PATH)
    connection.execute("PRAGMA foreign_keys = ON")

    try:
        cursor = connection.cursor()

        # Check prerequisite master data.
        warehouse_count = cursor.execute(
            "SELECT COUNT(*) FROM warehouses"
        ).fetchone()[0]

        zone_count = cursor.execute(
            "SELECT COUNT(*) FROM zones"
        ).fetchone()[0]

        requirement_count = cursor.execute(
            "SELECT COUNT(*) FROM storage_requirements"
        ).fetchone()[0]

        if warehouse_count != 10:
            raise RuntimeError(
                f"Expected 10 warehouses, found {warehouse_count}."
            )

        if zone_count != 100:
            raise RuntimeError(
                f"Expected 100 zones, found {zone_count}."
            )

        if requirement_count != 5:
            raise RuntimeError(
                f"Expected 5 requirements, found {requirement_count}."
            )

        existing_lots = cursor.execute(
            "SELECT COUNT(*) FROM inventory_lots"
        ).fetchone()[0]

        if existing_lots:
            raise RuntimeError(
                "Inventory lots already exist. "
                "Stopping to prevent duplicate records."
            )

        # Read storage requirement IDs.
        requirements = {
            product_type: requirement_id
            for requirement_id, product_type
            in cursor.execute(
                """
                SELECT requirement_id, product_type
                FROM storage_requirements
                """
            ).fetchall()
        }

        if set(requirements) != set(PRODUCT_ZONE_MAPPING):
            raise RuntimeError(
                "Storage requirement product types do not match "
                "the inventory generator configuration."
            )

        # Read compatible zones.
        zones_by_type = {}

        for zone_type in set(PRODUCT_ZONE_MAPPING.values()):
            zones_by_type[zone_type] = cursor.execute(
                """
                SELECT zone_id
                FROM zones
                WHERE zone_type = ?
                ORDER BY zone_id
                """,
                (zone_type,),
            ).fetchall()

            if not zones_by_type[zone_type]:
                raise RuntimeError(
                    f"No zones found for type {zone_type}."
                )

        product_types = list(PRODUCT_ZONE_MAPPING.keys())

        lot_records = []
        assignment_records = []

        for number in range(1, LOT_COUNT + 1):
            lot_id = f"LOT-{number:06d}"

            product_type = rng.choice(product_types)

            requirement_id = requirements[product_type]

            compatible_zone_type = PRODUCT_ZONE_MAPPING[product_type]

            zone_id = rng.choice(
                zones_by_type[compatible_zone_type]
            )[0]

            received_at = BASE_DATE + timedelta(
                days=rng.randint(0, 300),
                hours=rng.randint(0, 23),
                minutes=rng.randint(0, 59),
            )

            expiry_date = received_at + timedelta(
                days=rng.randint(30, 730)
            )

            quantity = rng.randint(10, 1000)

            product_name = (
                f"{product_type.replace('_', ' ').title()} "
                f"Batch {number:06d}"
            )

            lot_records.append(
                (
                    lot_id,
                    product_type,
                    product_name,
                    quantity,
                    "UNITS",
                    requirement_id,
                    format_timestamp(received_at),
                    format_timestamp(expiry_date),
                    "STORED",
                    format_timestamp(received_at),
                )
            )

            assignment_records.append(
                (
                    lot_id,
                    zone_id,
                    format_timestamp(received_at),
                    None,
                    "Initial compatible storage placement",
                    format_timestamp(received_at),
                )
            )

        # Insert inventory lots.
        cursor.executemany(
            """
            INSERT INTO inventory_lots (
                lot_id,
                product_type,
                product_name,
                quantity,
                unit_of_measure,
                requirement_id,
                received_at,
                expiry_date,
                lifecycle_state,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """,
            lot_records,
        )

        # Insert lot-zone assignments.
        cursor.executemany(
            """
            INSERT INTO lot_zone_assignments (
                lot_id,
                zone_id,
                assigned_at,
                released_at,
                assignment_reason,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            assignment_records,
        )

        connection.commit()

        print("Inventory generation successful!")
        print("Inventory lots:", len(lot_records))
        print("Lot-zone assignments:", len(assignment_records))

        print("\nLots by product type:")

        results = cursor.execute(
            """
            SELECT product_type, COUNT(*)
            FROM inventory_lots
            GROUP BY product_type
            ORDER BY product_type
            """
        ).fetchall()

        for product_type, count in results:
            print(f"{product_type}: {count:,}")

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


if __name__ == "__main__":
    generate_inventory()

"""Generate sensors and their zone assignments."""

import random
import sqlite3
from datetime import datetime, timedelta

from app.database.database import DATABASE_PATH
from scripts.phase10_generation_config import get_generator_seed

SEED = get_generator_seed('sensors')
SENSOR_TYPES = [
    "TEMP",
    "TEMP",
    "HUMIDITY",
    "OCCUPANCY",
    "MULTI",
]

MANUFACTURERS = [
    "Honeywell",
    "Siemens",
    "Bosch",
    "Sensirion",
    "Generic",
]


def generate_sensors():
    random.seed(SEED)

    connection = sqlite3.connect(DATABASE_PATH)
    connection.execute("PRAGMA foreign_keys = ON")

    try:
        cursor = connection.cursor()

        zone_count = cursor.execute(
            "SELECT COUNT(*) FROM zones"
        ).fetchone()[0]

        if zone_count != 100:
            raise RuntimeError(
                f"Expected 100 zones, found {zone_count}."
            )

        existing = cursor.execute(
            "SELECT COUNT(*) FROM sensors"
        ).fetchone()[0]

        if existing:
            raise RuntimeError(
                "Sensors already exist. Stopping to prevent duplicates."
            )

        zones = cursor.execute(
            "SELECT zone_id FROM zones ORDER BY zone_id"
        ).fetchall()

        now = datetime.now()
        sensor_records = []
        assignment_records = []

        sensor_number = 0

        for (zone_id,) in zones:
            for sensor_type in SENSOR_TYPES:
                sensor_number += 1

                sensor_id = f"SEN-{sensor_number:04d}"

                installation_date = now - timedelta(
                    days=random.randint(30, 365)
                )

                calibration_due = installation_date + timedelta(
                    days=180
                )

                sensor_records.append(
                    (
                        sensor_id,
                        f"{sensor_type} Sensor {sensor_number:04d}",
                        sensor_type,
                        random.choice(MANUFACTURERS),
                        installation_date.strftime("%Y-%m-%d %H:%M:%S"),
                        calibration_due.strftime("%Y-%m-%d %H:%M:%S"),
                        "ACTIVE",
                        installation_date.strftime("%Y-%m-%d %H:%M:%S"),
                    )
                )

                assignment_records.append(
                    (
                        sensor_id,
                        zone_id,
                        installation_date.strftime("%Y-%m-%d %H:%M:%S"),
                        None,
                        "Initial sensor installation",
                        installation_date.strftime("%Y-%m-%d %H:%M:%S"),
                    )
                )

        cursor.executemany(
            """
            INSERT INTO sensors (
                sensor_id,
                sensor_name,
                sensor_type,
                manufacturer,
                installation_date,
                calibration_due_date,
                lifecycle_state,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            """,
            sensor_records,
        )

        cursor.executemany(
            """
            INSERT INTO sensor_zone_assignments (
                sensor_id,
                zone_id,
                effective_from,
                effective_to,
                assignment_reason,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            assignment_records,
        )

        connection.commit()

        print("Sensor generation successful!")
        print("Total zones:", len(zones))
        print("Sensors per zone:", len(SENSOR_TYPES))
        print("Total sensors:", len(sensor_records))
        print("Sensor-zone assignments:", len(assignment_records))

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


if __name__ == "__main__":
    generate_sensors()

"""Generate deterministic, realistic sensor readings for Phase 4.

Run:
    python -m app.generator.sensor_reading_data
    python -m app.generator.sensor_reading_data --count 5000
    python -m app.generator.sensor_reading_data --replace

The default creates exactly 1,000,000 readings (2,000 per sensor when
the database contains the expected 500 sensors).
"""

import argparse
import math
import random
import sqlite3
from datetime import datetime, timedelta

from app.database.database import DATABASE_PATH
from scripts.phase10_generation_config import get_generator_seed

SEED = get_generator_seed('sensor_readings')
DEFAULT_READING_COUNT = 1_000_000
READING_INTERVAL_MINUTES = 5
SIMULATION_START = datetime(2026, 10, 22, 0, 0, 0)
BATCH_SIZE = 10_000

# Approximate normal operating temperatures by zone type (Celsius).
TEMPERATURE_BASELINES = {
    "FREEZER": -18.0,
    "COLD": 4.0,
    "HOT": 28.0,
    "AMBIENT": 22.0,
    "LOADING": 20.0,
}
HUMIDITY_BASELINES = {
    "FREEZER": 55.0,
    "COLD": 60.0,
    "HOT": 45.0,
    "AMBIENT": 48.0,
    "LOADING": 58.0,
}


def _clamp(value, lower, upper):
    return max(lower, min(upper, value))


def _load_sensors(connection):
    """Load each sensor with its currently assigned zone characteristics."""
    rows = connection.execute(
        """
        SELECT s.sensor_id, s.sensor_type, z.zone_id, z.zone_type,
               z.max_capacity
        FROM sensors AS s
        JOIN sensor_zone_assignments AS a
          ON a.sensor_id = s.sensor_id
         AND a.effective_to IS NULL
        JOIN zones AS z ON z.zone_id = a.zone_id
        WHERE s.lifecycle_state = 'ACTIVE'
        ORDER BY s.sensor_id
        """
    ).fetchall()

    if len(rows) != 500:
        raise RuntimeError(
            f"Expected 500 active sensors with current zone assignments; "
            f"found {len(rows)}. Check Phase 3 data before continuing."
        )

    sensor_ids = [row[0] for row in rows]
    if len(set(sensor_ids)) != len(sensor_ids):
        raise RuntimeError("A sensor has multiple current zone assignments.")

    return rows


def _make_reading(sensor, index, rng):
    sensor_id, sensor_type, _zone_id, zone_type, max_capacity = sensor
    timestamp = SIMULATION_START + timedelta(
        minutes=index * READING_INTERVAL_MINUTES
    )

    # Smooth daily cycle plus small random variation.
    day_angle = 2.0 * math.pi * (
        (timestamp.hour * 60 + timestamp.minute) / 1440.0
    )
    daily_wave = math.sin(day_angle - math.pi / 2.0)

    temperature = None
    humidity = None
    occupancy = None

    if sensor_type in ("TEMP", "MULTI"):
        baseline = TEMPERATURE_BASELINES.get(zone_type, 22.0)
        temperature = round(
            baseline + 0.7 * daily_wave + rng.gauss(0, 0.22), 2
        )

    if sensor_type in ("HUMIDITY", "MULTI"):
        baseline = HUMIDITY_BASELINES.get(zone_type, 50.0)
        humidity = round(
            _clamp(baseline + 2.0 * daily_wave + rng.gauss(0, 1.2), 0, 100),
            2,
        )

    if sensor_type in ("OCCUPANCY", "MULTI"):
        # Keep occupancy realistic and within the zone's declared capacity.
        capacity = max(1, int(max_capacity))
        typical_load = 0.16 + 0.08 * (1.0 + daily_wave)
        occupancy = int(
            _clamp(
                round(capacity * typical_load + rng.gauss(0, capacity * 0.025)),
                0,
                capacity,
            )
        )

    battery = round(
        _clamp(98.0 - (index * READING_INTERVAL_MINUTES / 1440.0) * 0.025
               + rng.uniform(-0.08, 0.08), 0, 100),
        2,
    )
    signal = round(_clamp(rng.gauss(-62.0, 7.0), -100, -25), 2)
    received_at = timestamp + timedelta(seconds=rng.randint(1, 45))

    reading_id = f"RDG-{sensor_id}-{index + 1:04d}"

    return (
        reading_id,
        sensor_id,
        timestamp.strftime("%Y-%m-%d %H:%M:%S"),
        temperature,
        humidity,
        occupancy,
        battery,
        signal,
        received_at.strftime("%Y-%m-%d %H:%M:%S"),
    )


def generate_sensor_readings(reading_count=DEFAULT_READING_COUNT, replace=False):
    if reading_count < 1:
        raise ValueError("reading_count must be at least 1.")

    rng = random.Random(SEED)
    connection = sqlite3.connect(DATABASE_PATH)
    connection.execute("PRAGMA foreign_keys = ON")

    try:
        existing = connection.execute(
            "SELECT COUNT(*) FROM sensor_readings"
        ).fetchone()[0]

        if existing and not replace:
            raise RuntimeError(
                f"sensor_readings already contains {existing:,} rows. "
                "Use --replace only if you intentionally want to regenerate it."
            )

        sensors = _load_sensors(connection)

        if replace:
            connection.execute("DELETE FROM sensor_readings")

        base, remainder = divmod(reading_count, len(sensors))
        batch = []
        inserted = 0

        for sensor_index, sensor in enumerate(sensors):
            sensor_reading_count = base + (1 if sensor_index < remainder else 0)

            for reading_index in range(sensor_reading_count):
                batch.append(_make_reading(sensor, reading_index, rng))

                if len(batch) >= BATCH_SIZE:
                    connection.executemany(
                        """
                        INSERT INTO sensor_readings (
                            reading_id, sensor_id, recorded_at,
                            temperature_c, humidity_pct, occupancy_count,
                            battery_pct, signal_strength, received_at
                        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
                        """,
                        batch,
                    )
                    inserted += len(batch)
                    batch.clear()

        if batch:
            connection.executemany(
                """
                INSERT INTO sensor_readings (
                    reading_id, sensor_id, recorded_at,
                    temperature_c, humidity_pct, occupancy_count,
                    battery_pct, signal_strength, received_at
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                batch,
            )
            inserted += len(batch)

        actual = connection.execute(
            "SELECT COUNT(*) FROM sensor_readings"
        ).fetchone()[0]
        if actual != reading_count:
            raise RuntimeError(
                f"Expected {reading_count:,} readings after insert; found {actual:,}."
            )

        connection.commit()

        print("Sensor reading generation successful!")
        print("Active sensors:", len(sensors))
        print("Readings inserted:", f"{inserted:,}")
        print("Total sensor_readings rows:", f"{actual:,}")
        print("Reading interval:", f"{READING_INTERVAL_MINUTES} minutes")
        print("Simulation start:", SIMULATION_START.isoformat(sep=" "))
        print("Random seed:", SEED)

    except Exception:
        connection.rollback()
        raise
    finally:
        connection.close()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--count", type=int, default=DEFAULT_READING_COUNT,
        help="Total readings to generate (default: 1,000,000).",
    )
    parser.add_argument(
        "--replace", action="store_true",
        help="Delete existing sensor readings before generating new ones.",
    )
    args = parser.parse_args()
    generate_sensor_readings(args.count, args.replace)


if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import json
import random
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]

MASTER_SEED = 20261022

# Keep generator seeds explicit so each data domain is reproducible.
GENERATOR_SEEDS = {
    "warehouses": MASTER_SEED,
    "zones": MASTER_SEED,
    "sensors": MASTER_SEED,
    "sensor_readings": MASTER_SEED,
    "inventory_lots": MASTER_SEED,
    "door_events": 20261023,
    "alarms": MASTER_SEED,
    "inspections": MASTER_SEED,
}

SCALE_PROFILES = {
    "starter": {
        "warehouses": 2,
        "zones": 20,
        "sensors": 100,
        "sensor_readings": 20000,
        "inventory_lots": 500,
    },
    "full": {
        "warehouses": 10,
        "zones": 100,
        "sensors": 500,
        "sensor_readings": 1000000,
        "inventory_lots": 10000,
    },
}


def get_scale_profile(name: str) -> dict:
    if name not in SCALE_PROFILES:
        raise ValueError(
            f"Unknown scale profile '{name}'. "
            f"Choose from: {', '.join(SCALE_PROFILES)}"
        )

    return SCALE_PROFILES[name].copy()


def get_generator_seed(generator_name: str) -> int:
    if generator_name not in GENERATOR_SEEDS:
        raise ValueError(
            f"Unknown generator: {generator_name}"
        )

    return GENERATOR_SEEDS[generator_name]


def create_rng(generator_name: str) -> random.Random:
    return random.Random(
        get_generator_seed(generator_name)
    )


def warehouse_id(number: int) -> str:
    return f"WH-{number:03d}"


def zone_id(warehouse_number: int, zone_number: int) -> str:
    return (
        f"ZN-{warehouse_number:03d}-{zone_number:02d}"
    )


def sensor_id(sensor_number: int) -> str:
    return f"SEN-{sensor_number:04d}"


def lot_id(lot_number: int) -> str:
    return f"LOT-2026-{lot_number:06d}"


def reading_id(sensor_identifier: str, timestamp: str) -> str:
    normalized = (
        timestamp.replace("-", "")
        .replace(":", "")
        .replace(" ", "T")
    )

    return f"RDG-{sensor_identifier}-{normalized}"


def build_configuration(profile_name: str) -> dict:
    return {
        "project": "Warehouse Risk Monitoring System",
        "master_seed": MASTER_SEED,
        "profile": profile_name,
        "scale": get_scale_profile(profile_name),
        "generator_seeds": GENERATOR_SEEDS,
        "id_formats": {
            "warehouse": "WH-{number:03d}",
            "zone": "ZN-{warehouse_number:03d}-{zone_number:02d}",
            "sensor": "SEN-{number:04d}",
            "inventory_lot": "LOT-2026-{number:06d}",
            "sensor_reading": "RDG-{sensor_id}-{normalized_timestamp}",
        },
        "reproducibility": {
            "fixed_seed": True,
            "stable_id_formats": True,

    "ground_truth_id",

def table_columns(connection, table_name):
    return {
        row["name"]
        for row in connection.execute(
            f'PRAGMA table_info("{table_name}")'
        )
    }


def first_column(columns, candidates):
    for candidate in candidates:
        if candidate in columns:
            return candidate
    return None


def parse_timestamp(value):
    if value is None:
        return None

    if isinstance(value, (int, float)):
        # Support Unix timestamps in seconds or milliseconds.
        number = float(value)
        if number > 10_000_000_000:
            number /= 1000
        try:
            return datetime.fromtimestamp(number, tz=timezone.utc)
        except (ValueError, OverflowError, OSError):
            return None

    value = str(value).strip()
    if not value:
        return None

    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))
        if parsed.tzinfo is None:
            parsed = parsed.replace(tzinfo=timezone.utc)
        return parsed.astimezone(timezone.utc)
    except ValueError:
        return None


def json_text(value):
    return json.dumps(value, sort_keys=True, default=str)


def add_candidate(
    candidates,
    issue_type,
    entity_type,
    entity_id,
    related_ids,
    evidence,
    source_tables,
):
    candidates.append({
        "issue_type": issue_type,
        "entity_type": entity_type,
        "entity_id": str(entity_id),
        "related_entity_ids": json_text(related_ids),
        "evidence": json_text(evidence),
        "label_status": "REVIEW_REQUIRED",
        "source_tables": json_text(source_tables),
    })


def detect_multi_zone_sensors(connection, candidates):
    table = "sensor_zone_assignments"
    existing = {
        row["name"]
        for row in connection.execute(
            f'PRAGMA table_info("{table}")'
        )
    } if table in get_tables(connection) else set()

    sensor_col = first_column(existing, ["sensor_id"])
    zone_col = first_column(existing, ["zone_id"])

    if not sensor_col or not zone_col:
        print("SKIP: Multi-zone sensor check (required columns unavailable)")
        return

    rows = connection.execute(
        f'SELECT "{sensor_col}", "{zone_col}" FROM "{table}"'
    ).fetchall()

    zones_by_sensor = defaultdict(set)
    for row in rows:
        if row[0] is not None and row[1] is not None:
            zones_by_sensor[str(row[0])].add(str(row[1]))

    count = 0
    for sensor_id, zone_ids in sorted(zones_by_sensor.items()):
        if len(zone_ids) > 1:
            add_candidate(
                candidates,
                "SENSOR_ASSIGNED_TO_MULTIPLE_ZONES",
                "sensor",
                sensor_id,
                {"zone_ids": sorted(zone_ids)},
                {
                    "distinct_zone_count": len(zone_ids),
                    "note": (
                        "Multiple zone assignments detected. Confirm "
                        "whether assignments overlap in effective time "
                        "or have an approved operational reason."
                    ),
                },
                [table],
            )
            count += 1

    print(f"Multi-zone sensor candidates: {count}")


def detect_alarm_without_lot_links(connection, candidates):
    tables = get_tables(connection)
    if "alarms" not in tables or "alarm_lots" not in tables:
        print("SKIP: Alarm-to-lot check (required tables unavailable)")
        return

    alarm_columns = table_columns(connection, "alarms")
    link_columns = table_columns(connection, "alarm_lots")

    alarm_id_col = first_column(alarm_columns, ["alarm_id", "id"])
    alarm_type_col = first_column(alarm_columns, ["alarm_type", "type"])
    link_alarm_col = first_column(link_columns, ["alarm_id"])

    if not alarm_id_col or not alarm_type_col or not link_alarm_col:
        print("SKIP: Alarm-to-lot check (required columns unavailable)")
        return

    linked_ids = {
        str(row[0])
        for row in connection.execute(
            f'SELECT DISTINCT "{link_alarm_col}" '
            f'FROM alarm_lots WHERE "{link_alarm_col}" IS NOT NULL'
        )
    }

    rows = connection.execute(
        f'SELECT "{alarm_id_col}", "{alarm_type_col}" '
        f'FROM alarms ORDER BY "{alarm_id_col}"'
    ).fetchall()

    count = 0
    expected_exceptions = 0

    for row in rows:
        alarm_id = str(row[0])
        alarm_type = str(row[1]).upper()

        # Door alarms are operational events. They do not necessarily
        # represent an inventory impact and must not be labelled as
        # missing inventory relationships by default.
        if alarm_type == "DOOR_OPEN":
            if alarm_id not in linked_ids:
                expected_exceptions += 1
            continue

        if alarm_id not in linked_ids:
            add_candidate(
                candidates,
                "ALARM_WITHOUT_INVENTORY_LOT_LINK",
                "alarm",
                alarm_id,
                {"lot_ids": []},
                {
                    "note": (
                        "Non-door alarm has no alarm_lots relationship. "
                        "Review whether inventory impact was expected."
                    )
                },
                ["alarms", "alarm_lots"],
            )
            count += 1

    print(f"Unlinked non-door alarm candidates: {count}")
    print(f"Unlinked DOOR_OPEN expected exceptions: {expected_exceptions}")


def detect_reading_gaps(connection, candidates):
    tables = get_tables(connection)
    if "sensor_readings" not in tables:
        print("SKIP: Reading-gap check (sensor_readings table unavailable)")
        return

    columns = table_columns(connection, "sensor_readings")
    sensor_col = first_column(columns, ["sensor_id"])
    timestamp_col = first_column(
        columns,
        ["recorded_at", "reading_timestamp", "timestamp", "reading_time"],
    )

    if not sensor_col or not timestamp_col:
        print("SKIP: Reading-gap check (required columns unavailable)")
        return

    rows = connection.execute(
        f'SELECT "{sensor_col}", "{timestamp_col}" '
        f'FROM sensor_readings '
        f'WHERE "{sensor_col}" IS NOT NULL '
        f'AND "{timestamp_col}" IS NOT NULL '
        f'ORDER BY "{sensor_col}", "{timestamp_col}"'
    ).fetchall()

    readings_by_sensor = defaultdict(list)
    for row in rows:
        parsed = parse_timestamp(row[1])
        if parsed:
            readings_by_sensor[str(row[0])].append(parsed)

    intervals = []
    sensor_intervals = {}

    for sensor_id, timestamps in readings_by_sensor.items():
        timestamps.sort()
        gaps = [
            (timestamps[index] - timestamps[index - 1]).total_seconds()
            for index in range(1, len(timestamps))
        ]
        gaps = [gap for gap in gaps if gap > 0]
        if gaps:
            sensor_intervals[sensor_id] = gaps
            intervals.extend(gaps)

    if not intervals:
        print("SKIP: Reading-gap check (no usable timestamp intervals)")
        return

    typical_interval = statistics.median(intervals)
    threshold = max(MIN_GAP_SECONDS, typical_interval * 3)

    count = 0
    for sensor_id, timestamps in readings_by_sensor.items():
        timestamps.sort()

        for index in range(1, len(timestamps)):
            gap_seconds = (
                timestamps[index] - timestamps[index - 1]
            ).total_seconds()

            if gap_seconds > threshold:
                add_candidate(
                    candidates,
                    "SENSOR_READING_GAP",
                    "sensor",
                    sensor_id,
                    {},
                    {
                        "gap_start": timestamps[index - 1].isoformat(),
                        "gap_end": timestamps[index].isoformat(),
                        "gap_seconds": round(gap_seconds, 2),
                        "threshold_seconds": round(threshold, 2),
                        "database_median_interval_seconds": round(
                            typical_interval, 2
                        ),
                    },
                    ["sensor_readings"],
                )
                count += 1

    print(f"Sensor reading-gap candidates: {count}")
    print(f"Calculated reading-gap threshold: {threshold:.2f} seconds")


def get_tables(connection):
    return {
        row[0]
        for row in connection.execute("""
            SELECT name
            FROM sqlite_master
            WHERE type = 'table'
              AND name NOT LIKE 'sqlite_%'
        """)
    }


def write_catalog():
    content = """# Phase 10 — Anomaly Catalog

These labels are generated from evidence in the source database.
They are stored separately from operational input data.

All automatically generated records initially use
`REVIEW_REQUIRED`. Review them against business rules before treating
them as confirmed ground truth.

| Issue type | Detection rule | Review required |
|---|---|---|
| `SENSOR_ASSIGNED_TO_MULTIPLE_ZONES` | A sensor has more than one distinct zone in sensor-zone assignments. | Confirm effective dates and approved multi-zone use. |
| `SENSOR_READING_GAP` | Consecutive readings for a sensor are separated by more than the calculated threshold. | Confirm expected sampling cadence and outage context. |
| `ALARM_WITHOUT_INVENTORY_LOT_LINK` | An alarm has no matching row in the alarm-lot bridge table. | Confirm whether the alarm should affect inventory. |

## Reading-gap threshold

The script calculates the median positive interval between readings
across sensors. The detection threshold is the greater of:

- 10 minutes
- Three times the calculated median interval

## Important

This catalog does not modify source tables, create alarms, or assign
remediation outcomes. It only writes a separate review dataset.
"""
    OUTPUT_CATALOG.write_text(content, encoding="utf-8")


def main():
    print("========== PHASE 10.3: GROUND TRUTH GENERATION ==========")

    if not DB_PATH.exists():
        raise SystemExit(f"Database not found: {DB_PATH}")

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    candidates = []

    connection = sqlite3.connect(DB_PATH)
    connection.row_factory = sqlite3.Row

    try:
        detect_multi_zone_sensors(connection, candidates)
        detect_reading_gaps(connection, candidates)
        detect_alarm_without_lot_links(connection, candidates)

        # Stable ordering makes IDs reproducible across identical runs.
        candidates.sort(
            key=lambda item: (
                item["issue_type"],
                item["entity_type"],
                item["entity_id"],
                item["evidence"],
            )
        )

        counters = defaultdict(int)
        output_rows = []

        for candidate in candidates:
            issue_type = candidate["issue_type"]
            counters[issue_type] += 1

            candidate["ground_truth_id"] = (
                f"GT-{issue_type}-{counters[issue_type]:06d}"
            )

            output_rows.append({
                "ground_truth_id": candidate["ground_truth_id"],
                **candidate,
            })

        with OUTPUT_CSV.open(
            "w", newline="", encoding="utf-8"
        ) as file:
            writer = csv.DictWriter(file, fieldnames=OUTPUT_FIELDS)
            writer.writeheader()
            writer.writerows(output_rows)

        write_catalog()

        table_counts = {}
        for table in sorted(get_tables(connection)):
            table_counts[table] = connection.execute(
                f'SELECT COUNT(*) FROM "{table}"'
            ).fetchone()[0]

    finally:
        connection.close()

    csv_hash = hashlib.sha256(OUTPUT_CSV.read_bytes()).hexdigest()

    manifest = {
        "phase": "10.3",
        "database": "data/warehouse_monitor.db",
        "ground_truth_file": "docs/phase10/ground_truth.csv",
        "anomaly_catalog": "docs/phase10/anomaly_catalog.md",
        "label_policy": (
            "Automatically detected candidates require manual review."
        ),
        "ground_truth_rows": len(output_rows),
        "issue_counts": dict(sorted(counters.items())),
        "source_table_counts": table_counts,
        "ground_truth_sha256": csv_hash,
    }

    OUTPUT_MANIFEST.write_text(
        json.dumps(manifest, indent=2, sort_keys=True),
        encoding="utf-8",
    )

    print("\n========== GENERATION SUMMARY ==========")
    print("Ground-truth rows:", len(output_rows))
    for issue_type, count in sorted(counters.items()):
        print(f"{issue_type}: {count}")

    print("\nCreated:")
    print(OUTPUT_CSV.relative_to(ROOT))
    print(OUTPUT_CATALOG.relative_to(ROOT))
    print(OUTPUT_MANIFEST.relative_to(ROOT))
    print("\nSource database was not modified.")


if __name__ == "__main__":
    main()

def quote_identifier(value: str) -> str:
    """Safely quote a SQLite identifier."""
    return '"' + value.replace('"', '""') + '"'


def load_table(conn, table_name):
    query = f"SELECT * FROM {quote_identifier(table_name)}"
    cursor = conn.execute(query)
    columns = [item[0] for item in cursor.description]
    return [
        dict(zip(columns, row))
        for row in cursor.fetchall()
    ]


def first_value(record, candidates):
    for column in candidates:
        if column in record and record[column] is not None:
            return record[column]
    return None


def matching_records(records, column, value):
    if value is None:
        return []

    return [
        record for record in records
        if record.get(column) == value
    ]


def main():
    if not DB_PATH.exists():
        raise FileNotFoundError(f"Database not found: {DB_PATH}")

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    conn = sqlite3.connect(
        f"file:{DB_PATH.resolve()}?mode=ro",
        uri=True
    )
    conn.row_factory = sqlite3.Row

    try:
        table_names = {
            row["name"]
            for row in conn.execute("""

            """)
        }

        required_tables = {
            "alarms",
            "alarm_lots",
            "inventory_lots",
            "inspections",
            "remediations",
        }

        missing = required_tables - table_names
        if missing:
            raise RuntimeError(
                f"Required tables missing: {sorted(missing)}"
            )

        alarms = load_table(conn, "alarms")
        alarm_lots = load_table(conn, "alarm_lots")
        inventory_lots = load_table(conn, "inventory_lots")
        inspections = load_table(conn, "inspections")
        remediations = load_table(conn, "remediations")

        lots_by_id = {
            lot.get("lot_id"): lot
            for lot in inventory_lots
            if lot.get("lot_id") is not None
        }

        rows = []

        for alarm in alarms:
            alarm_id = alarm.get("alarm_id")

            linked_lot_rows = matching_records(
                alarm_lots,
                "alarm_id",
                alarm_id
            )

            linked_lot_ids = sorted({
                link.get("lot_id")
                for link in linked_lot_rows
                if link.get("lot_id") is not None
            })

            alarm_inspections = matching_records(
                inspections,
                "alarm_id",
                alarm_id
            )

            alarm_remediations = matching_records(
                remediations,
                "alarm_id",
                alarm_id
            )

            # Also include remediation records linked through inspections.
            inspection_ids = {
                item.get("inspection_id")
                for item in alarm_inspections
                if item.get("inspection_id") is not None
            }

            for remediation in remediations:
                inspection_id = remediation.get("inspection_id")
                if inspection_id in inspection_ids:
                    if remediation not in alarm_remediations:
                        alarm_remediations.append(remediation)

            # Keep one report row per alarm-lot relationship.
            # Alarms without a linked lot still receive one row.
            report_lot_ids = linked_lot_ids or [None]

            for lot_id in report_lot_ids:
                lot = lots_by_id.get(lot_id, {})

                rows.append({
                    "alarm_id": alarm_id,
                    "warehouse_id": alarm.get("warehouse_id"),
                    "zone_id": alarm.get("zone_id"),
                    "sensor_id": alarm.get("sensor_id"),

                        alarm,

                        lot,

                    ),
                    "product_id": lot.get("product_id"),

from __future__ import annotations

import hashlib
import json
import sqlite3
from datetime import datetime, timezone
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
DB_PATH = ROOT / "data" / "warehouse_monitor.db"
OUTPUT_PATH = ROOT / "docs" / "phase10" / "generation_manifest.json"


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()

    with path.open("rb") as file:
        for chunk in iter(lambda: file.read(1024 * 1024), b""):
            digest.update(chunk)

    return digest.hexdigest()


def main():
    if not DB_PATH.exists():
        raise FileNotFoundError(DB_PATH)

    OUTPUT_PATH.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    conn = sqlite3.connect(
        f"file:{DB_PATH.resolve()}?mode=ro",
        uri=True
    )

    try:
        tables = conn.execute("""
            SELECT name
            FROM sqlite_master
            WHERE type = 'table'
              AND name NOT LIKE 'sqlite_%'
            ORDER BY name
        """).fetchall()

        table_counts = {}

        for (table_name,) in tables:
            # Table names originate from sqlite_master.
            quoted_name = '"' + table_name.replace('"', '""') + '"'

            count = conn.execute(
                f"SELECT COUNT(*) FROM {quoted_name}"
            ).fetchone()[0]

            table_counts[table_name] = count

        integrity = conn.execute(
            "PRAGMA integrity_check"
        ).fetchone()[0]

        foreign_key_errors = conn.execute(
            "PRAGMA foreign_key_check"
        ).fetchall()

        if integrity != "ok":
            raise RuntimeError(
                f"Database integrity failure: {integrity}"
            )

        if foreign_key_errors:
            raise RuntimeError(
                f"Foreign-key violations: {len(foreign_key_errors)}"
            )

    finally:
        conn.close()

    manifest = {
        "generated_at_utc": datetime.now(
            timezone.utc
        ).isoformat(),
        "database": "data/warehouse_monitor.db",
        "database_sha256": sha256_file(DB_PATH),
        "database_size_bytes": DB_PATH.stat().st_size,
        "master_seed": 20261022,
        "table_counts": table_counts,
        "total_tables": len(table_counts),
        "database_integrity": integrity,
        "foreign_key_violations": len(foreign_key_errors),
        "source_database_modified": False,
        "note": (
            "This is a baseline manifest. It does not claim "
            "that the existing generators have been unified "
            "or that two full regenerations have been compared."
        ),
    }

    with OUTPUT_PATH.open(
        "w",
        encoding="utf-8"
    ) as file:
        json.dump(manifest, file, indent=4)

    print("========== GENERATION BASELINE MANIFEST ==========")
    print("Tables:", len(table_counts))

    for table, count in table_counts.items():
        print(f"{table}: {count:,}")

    print("\nDatabase SHA-256:")
    print(manifest["database_sha256"])

    print("\nManifest:")
    print(OUTPUT_PATH)


if __name__ == "__main__":
    main()

            quoted = '"' + table.replace('"', '""') + '"'

            actual_count = conn.execute(
                f"SELECT COUNT(*) FROM {quoted}"
            ).fetchone()[0]

            check(
                actual_count == expected_count,
                f"{table}: expected {expected_count:,}, found {actual_count:,}"
